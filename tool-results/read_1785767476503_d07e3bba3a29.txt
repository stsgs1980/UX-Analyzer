    93→function ExportMarkdownButton({ result, designMd }: { result: AnalysisResult; designMd: string | null }) {
    94→  const handleExport = () => {
    95→    const md = buildMarkdownExport(result, designMd);
    96→    const blob = new Blob([md], { type: "text/markdown;charset=utf-8" });
    97→    const url = URL.createObjectURL(blob);
    98→    const a = document.createElement("a");
    99→    a.href = url;
   100→    const filename = result.teardown?.title
   101→      ? `${result.teardown.title.replace(/[^a-zA-Z0-9а-яА-Я]/g, "-").toLowerCase()}.md`
   102→      : `analysis-${Date.now()}.md`;
   103→    a.download = filename;
   104→    a.click();
   105→    URL.revokeObjectURL(url);
   106→    toast.success("Markdown отчёт скачан");
   107→  };
   108→  return (
   109→    <Button variant="outline" size="sm" onClick={handleExport}>
   110→      <FileDown className="h-3.5 w-3.5 mr-1.5" />
   111→      Экспорт Markdown
   112→    </Button>
   113→  );
   114→}

   544→function HeuristicTab({ data }: { data: AnalysisResult }) {
   545→  const he = data.heuristicEvaluation;
   546→  if (!he?.scores || he.scores.length === 0) {
   547→    return <EmptyState message="Данные эвристической оценки недоступны" />;
   548→  }
   549→
   550→  const avg = he.averageScore ?? 0;
   551→
   552→  const scoreColor = (s: number) => {
   553→    if (s >= 3.5) return "text-emerald-600 dark:text-emerald-400";
   554→    if (s >= 2.5) return "text-amber-600 dark:text-amber-400";
   555→    if (s >= 1.5) return "text-orange-600 dark:text-orange-400";
   556→    return "text-red-600 dark:text-red-400";
   557→  };
   558→
   559→  const scoreBg = (s: number) => {
   560→    if (s >= 3.5) return "bg-emerald-500";
   561→    if (s >= 2.5) return "bg-amber-500";
   562→    if (s >= 1.5) return "bg-orange-500";
   563→    return "bg-red-500";
   564→  };
   565→
   566→  return (
   567→    <div className="space-y-6">
   568→      {/* Average score */}
   569→      <div className="border-l-2 border-l-primary/40 pl-4 py-3">
   570→        <div className="flex flex-col sm:flex-row items-center gap-4">
   571→          <div className="flex items-baseline gap-1">
   572→            <span className={`text-4xl font-bold ${scoreColor(avg)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   573→              {avg.toFixed(1)}
   574→            </span>
   575→            <span className="text-lg text-muted-foreground">/4.0</span>
   576→          </div>
   577→          <div className="text-center sm:text-left">
   578→            <p className="font-semibold text-lg">Общий балл</p>
   579→            <p className="text-sm text-muted-foreground mt-1">
   580→              {he.verdict || "Без вердикта"}
   581→            </p>
   582→          </div>
   583→        </div>
   584→      </div>
   585→
   586→      {/* Individual scores */}
   587→      <div className="space-y-3">
   588→        {he.scores.map((item, i) => (
   589→          <div key={i} className="border-b border-white/5 pb-3 pt-3">
   590→            <div className="flex items-start justify-between gap-4">
   591→              <div className="flex-1 space-y-1">
   592→                <p className="text-sm font-medium">{item.heuristic}</p>
   593→                {item.observations && (
   594→                  <p className="text-xs text-muted-foreground">{item.observations}</p>
   595→                )}
   596→                {item.recommendation && (
   597→                    <p className="text-xs text-muted-foreground italic mt-1">
   598→                      {item.recommendation}
   599→                    </p>
   600→                )}
   601→              </div>
   602→              <div className="flex flex-col items-center gap-1 shrink-0">
   603→                <span className={`text-lg font-bold ${scoreColor(item.score)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   604→                  {item.score}
   605→                </span>
   606→                <div className="flex gap-0.5">
   607→                  {[0, 1, 2, 3].map((dot) => (
   608→                    <span
   609→                      key={dot}
   610→                      className={`inline-block h-1.5 w-5 ${
   611→                        dot < item.score ? scoreBg(item.score) : "bg-muted"
   612→                      }`}
   613→                    />
   614→                  ))}
   615→                </div>
   616→              </div>
   617→            </div>
   618→          </div>
   619→        ))}
   620→      </div>
   621→    </div>
   622→  );
   623→}

   625→/* ─── EMPTY STATE ─── */
   626→function EmptyState({ message }: { message: string }) {
   627→  return (
   628→    <div className="py-12 text-center">
   629→      <FileText className="h-8 w-8 text-muted-foreground/30 mx-auto mb-3" aria-hidden="true" />
   630→      <p className="text-muted-foreground">{message}</p>
   631→    </div>
   632→  );
   633→}

   635→/* ─── META INFO ─── */
   636→function MetaInfo({ data }: { data: AnalysisResult }) {
   637→  const m = data.meta;
   638→  if (!m) return null;
   639→
   640→  return (
   641→    <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   642→      {m.dataSources?.map((ds) => (
   643→        <Badge key={ds} variant="outline" className="text-xs">
   644→          {ds}
   645→        </Badge>
   646→      ))}
   647→      {m.confidence && <ConfidenceBadge confidence={m.confidence} />}
   648→      {m.caveats?.map((c, i) => (
   649→        <span key={i} className="text-xs text-muted-foreground italic">
   650→          {c}
   651→        </span>
   652→      ))}
   653→    </div>
   654→  );
   655→}
