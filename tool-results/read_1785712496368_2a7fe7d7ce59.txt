    96→function ExportMarkdownButton({ result, designMd }: { result: AnalysisResult; designMd: string | null }) {
    97→  const handleExport = () => {
    98→    const md = buildMarkdownExport(result, designMd);
    99→    const blob = new Blob([md], { type: "text/markdown;charset=utf-8" });
   100→    const url = URL.createObjectURL(blob);
   101→    const a = document.createElement("a");
   102→    a.href = url;
   103→    const filename = result.teardown?.title
   104→      ? `${result.teardown.title.replace(/[^a-zA-Z0-9а-яА-Я]/g, "-").toLowerCase()}.md`
   105→      : `analysis-${Date.now()}.md`;
   106→    a.download = filename;
   107→    a.click();
   108→    URL.revokeObjectURL(url);
   109→    toast.success("Markdown отчёт скачан");
   110→  };
   111→  return (
   112→    <Button variant="outline" size="sm" onClick={handleExport}>
   113→      <FileDown className="h-3.5 w-3.5 mr-1.5" />
   114→      Экспорт Markdown
   115→    </Button>
   116→  );
   117→}

   544→}

   546→/* ─── HEURISTIC EVALUATION TAB ─── */
   547→function HeuristicTab({ data }: { data: AnalysisResult }) {
   548→  const he = data.heuristicEvaluation;
   549→  if (!he?.scores || he.scores.length === 0) {
   550→    return <EmptyState message="Данные эвристической оценки недоступны" />;
   551→  }
   552→
   553→  const avg = he.averageScore ?? 0;
   554→
   555→  const scoreColor = (s: number) => {
   556→    if (s >= 3.5) return "text-emerald-600 dark:text-emerald-400";
   557→    if (s >= 2.5) return "text-amber-600 dark:text-amber-400";
   558→    if (s >= 1.5) return "text-orange-600 dark:text-orange-400";
   559→    return "text-red-600 dark:text-red-400";
   560→  };
   561→
   562→  const scoreBg = (s: number) => {
   563→    if (s >= 3.5) return "bg-emerald-500";
   564→    if (s >= 2.5) return "bg-amber-500";
   565→    if (s >= 1.5) return "bg-orange-500";
   566→    return "bg-red-500";
   567→  };
   568→
   569→  return (
   570→    <div className="space-y-6">
   571→      {/* Average score */}
   572→      <div className="border-l-2 border-l-primary/40 pl-4 py-3">
   573→        <div className="flex flex-col sm:flex-row items-center gap-4">
   574→          <div className="flex items-baseline gap-1">
   575→            <span className={`text-4xl font-bold ${scoreColor(avg)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   576→              {avg.toFixed(1)}
   577→            </span>
   578→            <span className="text-lg text-muted-foreground">/4.0</span>
   579→          </div>
   580→          <div className="text-center sm:text-left">
   581→            <p className="font-semibold text-lg">Общий балл</p>
   582→            <p className="text-sm text-muted-foreground mt-1">
   583→              {he.verdict || "Без вердикта"}
   584→            </p>
   585→          </div>
   586→        </div>
   587→      </div>
   588→
   589→      {/* Individual scores */}
   590→      <div className="space-y-3">
   591→        {he.scores.map((item, i) => (
   592→          <div key={i} className="border-b border-white/5 pb-3 pt-3">
   593→            <div className="flex items-start justify-between gap-4">
   594→              <div className="flex-1 space-y-1">
   595→                <p className="text-sm font-medium">{item.heuristic}</p>
   596→                {item.observations && (
   597→                  <p className="text-xs text-muted-foreground">{item.observations}</p>
   598→                )}
   599→                {item.recommendation && (
   600→                    <p className="text-xs text-muted-foreground italic mt-1">
   601→                      {item.recommendation}
   602→                    </p>
   603→                )}
   604→              </div>
   605→              <div className="flex flex-col items-center gap-1 shrink-0">
   606→                <span className={`text-lg font-bold ${scoreColor(item.score)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   607→                  {item.score}
   608→                </span>
   609→                <div className="flex gap-0.5">
   610→                  {[0, 1, 2, 3].map((dot) => (
   611→                    <span
   612→                      key={dot}
   613→                      className={`inline-block h-1.5 w-5 ${
   614→                        dot < item.score ? scoreBg(item.score) : "bg-muted"
   615→                      }`}
   616→                    />
   617→                  ))}
   618→                </div>
   619→              </div>
   620→            </div>
   621→          </div>
   622→        ))}
   623→      </div>
   624→    </div>
   625→  );
   626→}

   628→/* ─── EMPTY STATE ─── */
   629→function EmptyState({ message }: { message: string }) {
   630→  return (
   631→    <div className="py-12 text-center">
   632→      <FileText className="h-8 w-8 text-muted-foreground/30 mx-auto mb-3" aria-hidden="true" />
   633→      <p className="text-muted-foreground">{message}</p>
   634→    </div>
   635→  );
   636→}

   638→/* ─── META INFO ─── */
   639→function MetaInfo({ data }: { data: AnalysisResult }) {
   640→  const m = data.meta;
   641→  if (!m) return null;
   642→
   643→  return (
   644→    <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   645→      {m.dataSources?.map((ds) => (
   646→        <Badge key={ds} variant="outline" className="text-xs">
   647→          {ds}
   648→        </Badge>
   649→      ))}
   650→      {m.confidence && <ConfidenceBadge confidence={m.confidence} />}
   651→      {m.caveats?.map((c, i) => (
   652→        <span key={i} className="text-xs text-muted-foreground italic">
   653→          {c}
   654→        </span>
   655→      ))}
   656→    </div>
   657→  );
   658→}
