    94→function ExportMarkdownButton({ result, designMd }: { result: AnalysisResult; designMd: string | null }) {
    95→  const handleExport = () => {
    96→    const md = buildMarkdownExport(result, designMd);
    97→    const blob = new Blob([md], { type: "text/markdown;charset=utf-8" });
    98→    const url = URL.createObjectURL(blob);
    99→    const a = document.createElement("a");
   100→    a.href = url;
   101→    const filename = result.teardown?.title
   102→      ? `${result.teardown.title.replace(/[^a-zA-Z0-9а-яА-Я]/g, "-").toLowerCase()}.md`
   103→      : `analysis-${Date.now()}.md`;
   104→    a.download = filename;
   105→    a.click();
   106→    URL.revokeObjectURL(url);
   107→    toast.success("Markdown отчёт скачан");
   108→  };
   109→  return (
   110→    <Button variant="outline" size="sm" onClick={handleExport}>
   111→      <FileDown className="h-3.5 w-3.5 mr-1.5" />
   112→      Экспорт Markdown
   113→    </Button>
   114→  );
   115→}

   544→/* ─── HEURISTIC EVALUATION TAB ─── */
   545→function HeuristicTab({ data }: { data: AnalysisResult }) {
   546→  const he = data.heuristicEvaluation;
   547→  if (!he?.scores || he.scores.length === 0) {
   548→    return <EmptyState message="Данные эвристической оценки недоступны" />;
   549→  }
   550→
   551→  const avg = he.averageScore ?? 0;
   552→
   553→  const scoreColor = (s: number) => {
   554→    if (s >= 3.5) return "text-emerald-600 dark:text-emerald-400";
   555→    if (s >= 2.5) return "text-amber-600 dark:text-amber-400";
   556→    if (s >= 1.5) return "text-orange-600 dark:text-orange-400";
   557→    return "text-red-600 dark:text-red-400";
   558→  };
   559→
   560→  const scoreBg = (s: number) => {
   561→    if (s >= 3.5) return "bg-emerald-500";
   562→    if (s >= 2.5) return "bg-amber-500";
   563→    if (s >= 1.5) return "bg-orange-500";
   564→    return "bg-red-500";
   565→  };
   566→
   567→  return (
   568→    <div className="space-y-6">
   569→      {/* Average score */}
   570→      <div className="border-l-2 border-l-primary/40 pl-4 py-3">
   571→        <div className="flex flex-col sm:flex-row items-center gap-4">
   572→          <div className="flex items-baseline gap-1">
   573→            <span className={`text-4xl font-bold ${scoreColor(avg)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   574→              {avg.toFixed(1)}
   575→            </span>
   576→            <span className="text-lg text-muted-foreground">/4.0</span>
   577→          </div>
   578→          <div className="text-center sm:text-left">
   579→            <p className="font-semibold text-lg">Общий балл</p>
   580→            <p className="text-sm text-muted-foreground mt-1">
   581→              {he.verdict || "Без вердикта"}
   582→            </p>
   583→          </div>
   584→        </div>
   585→      </div>
   586→
   587→      {/* Individual scores */}
   588→      <div className="space-y-3">
   589→        {he.scores.map((item, i) => (
   590→          <div key={i} className="border-b border-white/5 pb-3 pt-3">
   591→            <div className="flex items-start justify-between gap-4">
   592→              <div className="flex-1 space-y-1">
   593→                <p className="text-sm font-medium">{item.heuristic}</p>
   594→                {item.observations && (
   595→                  <p className="text-xs text-muted-foreground">{item.observations}</p>
   596→                )}
   597→                {item.recommendation && (
   598→                    <p className="text-xs text-muted-foreground italic mt-1">
   599→                      {item.recommendation}
   600→                    </p>
   601→                )}
   602→              </div>
   603→              <div className="flex flex-col items-center gap-1 shrink-0">
   604→                <span className={`text-lg font-bold ${scoreColor(item.score)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   605→                  {item.score}
   606→                </span>
   607→                <div className="flex gap-0.5">
   608→                  {[0, 1, 2, 3].map((dot) => (
   609→                    <span
   610→                      key={dot}
   611→                      className={`inline-block h-1.5 w-5 ${
   612→                        dot < item.score ? scoreBg(item.score) : "bg-muted"
   613→                      }`}
   614→                    />
   615→                  ))}
   616→                </div>
   617→              </div>
   618→            </div>
   619→          </div>
   620→        ))}
   621→      </div>
   622→    </div>
   623→  );
   624→}

   626→/* ─── EMPTY STATE ─── */
   627→function EmptyState({ message }: { message: string }) {
   628→  return (
   629→    <div className="py-12 text-center">
   630→      <FileText className="h-8 w-8 text-muted-foreground/30 mx-auto mb-3" aria-hidden="true" />
   631→      <p className="text-muted-foreground">{message}</p>
   632→    </div>
   633→  );
   634→}

   636→/* ─── META INFO ─── */
   637→function MetaInfo({ data }: { data: AnalysisResult }) {
   638→  const m = data.meta;
   639→  if (!m) return null;
   640→
   641→  return (
   642→    <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   643→      {m.dataSources?.map((ds) => (
   644→        <Badge key={ds} variant="outline" className="text-xs">
   645→          {ds}
   646→        </Badge>
   647→      ))}
   648→      {m.confidence && <ConfidenceBadge confidence={m.confidence} />}
   649→      {m.caveats?.map((c, i) => (
   650→        <span key={i} className="text-xs text-muted-foreground italic">
   651→          {c}
   652→        </span>
   653→      ))}
   654→    </div>
   655→  );
   656→}
