   104→   104→    a.click();
   105→   105→    URL.revokeObjectURL(url);
   106→   106→    toast.success("Markdown отчёт скачан");
   107→   107→  };

   544→   544→function HeuristicTab({ data }: { data: AnalysisResult }) {
   545→   545→  const he = data.heuristicEvaluation;
   546→   546→  if (!he?.scores || he.scores.length === 0) {
   547→   547→    return <EmptyState message="Данные эвристической оценки недоступны" />;
   548→   548→  }
   549→   549→
   550→   550→  const avg = he.averageScore ?? 0;
   551→   551→
   552→   552→  const scoreColor = (s: number) => {
   553→   553→    if (s >= 3.5) return "text-emerald-600 dark:text-emerald-400";
   554→   554→    if (s >= 2.5) return "text-amber-600 dark:text-amber-400";
   555→   555→    if (s >= 1.5) return "text-orange-600 dark:text-orange-400";
   556→   556→    return "text-red-600 dark:text-red-400";
   557→   557→  };
   558→   558→
   559→   559→  const scoreBg = (s: number) => {
   560→   560→    if (s >= 3.5) return "bg-emerald-500";
   561→   561→    if (s >= 2.5) return "bg-amber-500";
   562→   562→    if (s >= 1.5) return "bg-orange-500";
   563→   563→    return "bg-red-500";
   564→   564→  };
   565→   565→
   566→   566→  return (
   567→   567→    <div className="space-y-6">
   568→   568→      {/* Average score */}
   569→   569→      <div className="border-l-2 border-l-primary/40 pl-4 py-3">
   570→   570→        <div className="flex flex-col sm:flex-row items-center gap-4">
   571→   571→          <div className="flex items-baseline gap-1">
   572→   572→            <span className={`text-4xl font-bold ${scoreColor(avg)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   573→   573→              {avg.toFixed(1)}
   574→   574→            </span>
   575→   575→            <span className="text-lg text-muted-foreground">/4.0</span>
   576→   576→          </div>
   577→   577→          <div className="text-center sm:text-left">
   578→   578→            <p className="font-semibold text-lg">Общий балл</p>
   579→   579→            <p className="text-sm text-muted-foreground mt-1">
   580→   580→              {he.verdict || "Без вердикта"}
   581→   581→            </p>
   582→   582→          </div>
   583→   583→        </div>
   584→   584→      </div>
   585→   585→
   586→   586→      {/* Individual scores */}
   587→   587→      <div className="space-y-3">
   588→   588→        {he.scores.map((item, i) => (
   589→   589→          <div key={i} className="border-b border-white/5 pb-3 pt-3">
   590→   590→            <div className="flex items-start justify-between gap-4">
   591→   591→              <div className="flex-1 space-y-1">
   592→   592→                <p className="text-sm font-medium">{item.heuristic}</p>
   593→   593→                {item.observations && (
   594→   594→                  <p className="text-xs text-muted-foreground">{item.observations}</p>
   595→   595→                )}
   596→   596→                {item.recommendation && (
   597→   597→                    <p className="text-xs text-muted-foreground italic mt-1">
   598→   598→                      {item.recommendation}
   599→   599→                    </p>
   600→   600→                )}
   601→   601→              </div>
   602→   602→              <div className="flex flex-col items-center gap-1 shrink-0">
   603→   603→                <span className={`text-lg font-bold ${scoreColor(item.score)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   604→   604→                  {item.score}
   605→   605→                </span>
   606→   606→                <div className="flex gap-0.5">
   607→   607→                  {[0, 1, 2, 3].map((dot) => (
   608→   608→                    <span
   609→   609→                      key={dot}
   610→   610→                      className={`inline-block h-1.5 w-5 ${
   611→   611→                        dot < item.score ? scoreBg(item.score) : "bg-muted"
   612→   612→                      }`}
   613→   613→                    />
   614→   614→                  ))}
   615→   615→                </div>
   616→   616→              </div>
   617→   617→            </div>
   618→   618→          </div>
   619→   619→        ))}
   620→   620→      </div>
   621→   621→    </div>
   622→   622→  );
   623→   623→}
