   518→   518→/* ─── HEURISTIC EVALUATION TAB ─── */
   519→   519→function HeuristicTab({ data }: { data: AnalysisResult }) {
   520→   520→  const he = data.heuristicEvaluation;
   521→   521→  if (!he?.scores || he.scores.length === 0) {
   522→   522→    return <EmptyState message="Данные эвристической оценки недоступны" />;
   523→   523→  }
   524→   524→
   525→   525→  const avg = he.averageScore ?? 0;
   526→   526→
   527→   527→  const scoreColor = (s: number) => {
   528→   528→    if (s >= 3.5) return "text-emerald-600 dark:text-emerald-400";
   529→   529→    if (s >= 2.5) return "text-amber-600 dark:text-amber-400";
   530→   530→    if (s >= 1.5) return "text-orange-600 dark:text-orange-400";
   531→   531→    return "text-red-600 dark:text-red-400";
   532→   532→  };
   533→   533→
   534→   534→  const scoreBg = (s: number) => {
   535→   535→    if (s >= 3.5) return "bg-emerald-500";
   536→   536→    if (s >= 2.5) return "bg-amber-500";
   537→   537→    if (s >= 1.5) return "bg-orange-500";
   538→   538→    return "bg-red-500";
   539→   539→  };
   540→   540→
   541→   541→  return (
   542→   542→    <div className="space-y-6">
   543→   543→      {/* Average score */}
   544→   544→      <div className="border-l-2 border-l-primary/40 pl-4 py-3">
   545→   545→        <div className="flex flex-col sm:flex-row items-center gap-4">
   546→   546→          <div className="flex items-baseline gap-1">
   547→   547→            <span className={`text-4xl font-bold ${scoreColor(avg)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   548→   548→              {avg.toFixed(1)}
   549→   549→            </span>
   550→   550→            <span className="text-lg text-muted-foreground">/4.0</span>
   551→   551→          </div>
   552→   552→          <div className="text-center sm:text-left">
   553→   553→            <p className="font-semibold text-lg">Общий балл</p>
   554→   554→            <p className="text-sm text-muted-foreground mt-1">
   555→   555→              {he.verdict || "Без вердикта"}
   556→   556→            </p>
   557→   557→          </div>
   558→   558→        </div>
   559→   559→      </div>
   560→   560→
   561→   561→      {/* Individual scores */}
   562→   562→      <div className="space-y-3">
   563→   563→        {he.scores.map((item, i) => (
   564→   564→          <div key={i} className="border-b border-white/5 pb-3 pt-3">
   565→   565→            <div className="flex items-start justify-between gap-4">
   566→   566→              <div className="flex-1 space-y-1">
   567→   567→                <p className="text-sm font-medium">{item.heuristic}</p>
   568→   568→                {item.observations && (
   569→   569→                  <p className="text-xs text-muted-foreground">{item.observations}</p>
   570→   570→                )}
   571→   571→                {item.recommendation && (
   572→   572→                    <p className="text-xs text-muted-foreground italic mt-1">
   573→   573→                      {item.recommendation}
   574→   574→                    </p>
   575→   575→                )}
   576→   576→              </div>
   577→   577→              <div className="flex flex-col items-center gap-1 shrink-0">
   578→   578→                <span className={`text-lg font-bold ${scoreColor(item.score)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   579→   579→                  {item.score}
   580→   580→                </span>
   581→   581→                <div className="flex gap-0.5">
   582→   582→                  {[0, 1, 2, 3].map((dot) => (
   583→   583→                    <span
   584→   584→                      key={dot}
   585→   585→                      className={`inline-block h-1.5 w-5 ${
   586→   586→                        dot < item.score ? scoreBg(item.score) : "bg-muted"
   587→   587→                      }`}
   588→   588→                    />
   589→   589→                  ))}
   590→   590→                </div>
   591→   591→              </div>
   592→   592→            </div>
   593→   593→          </div>
   594→   594→        ))}
   595→   595→      </div>
   596→   596→    </div>
   597→   597→  );
   598→   598→}
