    91→function DownloadButton({ data }: { data: unknown }) {
    92→  const handleDownload = () => {
    93→    const blob = new Blob([JSON.stringify(data, null, 2)], {
    94→      type: "application/json",
    95→    });
    96→    const url = URL.createObjectURL(blob);
    97→    const a = document.createElement("a");
    98→    a.href = url;
    99→    a.download = `analysis-${Date.now()}.json`;
   100→    a.click();
   101→    URL.revokeObjectURL(url);
   102→  };
   103→  return (
   104→    <Button variant="outline" size="sm" onClick={handleDownload}>
   105→      <Download className="h-3.5 w-3.5 mr-1.5" />
   106→      Скачать JSON
   107→    </Button>

   522→    return <EmptyState message="Данные эвристической оценки недоступны" />;
   523→  }
   524→
   525→  const avg = he.averageScore ?? 0;
   526→
   527→  const scoreColor = (s: number) => {
   528→    if (s >= 3.5) return "text-emerald-600 dark:text-emerald-400";
   529→    if (s >= 2.5) return "text-amber-600 dark:text-amber-400";
   530→    if (s >= 1.5) return "text-orange-600 dark:text-orange-400";
   531→    return "text-red-600 dark:text-red-400";
   532→  };
   533→
   534→  const scoreBg = (s: number) => {
   535→    if (s >= 3.5) return "bg-emerald-500";
   536→    if (s >= 2.5) return "bg-amber-500";
   537→    if (s >= 1.5) return "bg-orange-500";
   538→    return "bg-red-500";
   539→  };
   540→
   541→  return (
   542→    <div className="space-y-6">
   543→      {/* Average score */}
   544→      <div className="border-l-2 border-l-primary/40 pl-4 py-3">
   545→        <div className="flex flex-col sm:flex-row items-center gap-4">
   546→          <div className="flex items-baseline gap-1">
   547→            <span className={`text-4xl font-bold ${scoreColor(avg)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   548→              {avg.toFixed(1)}
   549→            </span>
   550→            <span className="text-lg text-muted-foreground">/4.0</span>
   551→          </div>

   555→              {he.verdict || "Без вердикта"}
   556→            </p>
   557→          </div>
   558→        </div>
   559→      </div>
   560→
   561→      {/* Individual scores */}
   562→      <div className="space-y-3">
   563→        {he.scores.map((item, i) => (
   564→          <div key={i} className="border-b border-white/5 pb-3 pt-3">
   565→            <div className="flex items-start justify-between gap-4">
   566→              <div className="flex-1 space-y-1">
   567→                <p className="text-sm font-medium">{item.heuristic}</p>
   568→                {item.observations && (
   569→                  <p className="text-xs text-muted-foreground">{item.observations}</p>
   570→                )}
   571→                {item.recommendation && (
   572→                    <p className="text-xs text-muted-foreground italic mt-1">
   573→                      {item.recommendation}
   574→                    </p>
   575→                )}
   576→              </div>
   577→              <div className="flex flex-col items-center gap-1 shrink-0">
   578→                <span className={`text-lg font-bold ${scoreColor(item.score)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   579→                  {item.score}
   580→                </span>
   581→                <div className="flex gap-0.5">
   582→                  {[0, 1, 2, 3].map((dot) => (
   583→                    <span
   584→                      key={dot}
   585→                      className={`inline-block h-1.5 w-5 ${
   586→                        dot < item.score ? scoreBg(item.score) : "bg-muted"
   587→                      }`}
   588→                    />
   589→                  ))}
   590→                </div>
   591→              </div>
   592→            </div>
   593→          </div>
   594→        ))}
   595→      </div>
   596→    </div>
   597→  );
   598→}

   611→function MetaInfo({ data }: { data: AnalysisResult }) {
   612→  const m = data.meta;
   613→  if (!m) return null;
   614→
   615→  return (
   616→    <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   617→      {m.dataSources?.map((ds) => (
   618→        <Badge key={ds} variant="outline" className="text-xs">
   619→          {ds}
   620→        </Badge>
   621→      ))}
   622→      {m.confidence && <ConfidenceBadge confidence={m.confidence} />}
   623→      {m.caveats?.map((c, i) => (
   624→        <span key={i} className="text-xs text-muted-foreground italic">
   625→          {c}
   626→        </span>
   627→      ))}
   628→    </div>
   629→  );
   630→}
