   107→   107→    a.click();
   108→   108→    URL.revokeObjectURL(url);

   110→   110→  };

   547→   547→function HeuristicTab({ data }: { data: AnalysisResult }) {
   548→   548→  const he = data.heuristicEvaluation;
   549→   549→  if (!he?.scores || he.scores.length === 0) {
   550→   550→    return <EmptyState message="Данные эвристической оценки недоступны" />;
   551→   551→  }
   552→   552→
   553→   553→  const avg = he.averageScore ?? 0;
   554→   554→
   555→   555→  const scoreColor = (s: number) => {
   556→   556→    if (s >= 3.5) return "text-emerald-600 dark:text-emerald-400";
   557→   557→    if (s >= 2.5) return "text-amber-600 dark:text-amber-400";
   558→   558→    if (s >= 1.5) return "text-orange-600 dark:text-orange-400";
   559→   559→    return "text-red-600 dark:text-red-400";
   560→   560→  };
   561→   561→
   562→   562→  const scoreBg = (s: number) => {
   563→   563→    if (s >= 3.5) return "bg-emerald-500";
   564→   564→    if (s >= 2.5) return "bg-amber-500";
   565→   565→    if (s >= 1.5) return "bg-orange-500";
   566→   566→    return "bg-red-500";
   567→   567→  };
   568→   568→
   569→   569→  return (
   570→   570→    <div className="space-y-6">
   571→   571→      {/* Average score */}
   572→   572→      <div className="border-l-2 border-l-primary/40 pl-4 py-3">
   573→   573→        <div className="flex flex-col sm:flex-row items-center gap-4">
   574→   574→          <div className="flex items-baseline gap-1">
   575→   575→            <span className={`text-4xl font-bold ${scoreColor(avg)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   576→   576→              {avg.toFixed(1)}
   577→   577→            </span>
   578→   578→            <span className="text-lg text-muted-foreground">/4.0</span>
   579→   579→          </div>
   580→   580→          <div className="text-center sm:text-left">
   581→   581→            <p className="font-semibold text-lg">Общий балл</p>
   582→   582→            <p className="text-sm text-muted-foreground mt-1">
   583→   583→              {he.verdict || "Без вердикта"}
   584→   584→            </p>
   585→   585→          </div>
   586→   586→        </div>
   587→   587→      </div>
   588→   588→
   589→   589→      {/* Individual scores */}
   590→   590→      <div className="space-y-3">
   591→   591→        {he.scores.map((item, i) => (
   592→   592→          <div key={i} className="border-b border-white/5 pb-3 pt-3">
   593→   593→            <div className="flex items-start justify-between gap-4">
   594→   594→              <div className="flex-1 space-y-1">
   595→   595→                <p className="text-sm font-medium">{item.heuristic}</p>
   596→   596→                {item.observations && (
   597→   597→                  <p className="text-xs text-muted-foreground">{item.observations}</p>
   598→   598→                )}
   599→   599→                {item.recommendation && (
   600→   600→                    <p className="text-xs text-muted-foreground italic mt-1">
   601→   601→                      {item.recommendation}
   602→   602→                    </p>
   603→   603→                )}
   604→   604→              </div>
   605→   605→              <div className="flex flex-col items-center gap-1 shrink-0">
   606→   606→                <span className={`text-lg font-bold ${scoreColor(item.score)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   607→   607→                  {item.score}
   608→   608→                </span>
   609→   609→                <div className="flex gap-0.5">
   610→   610→                  {[0, 1, 2, 3].map((dot) => (

   612→   612→                      key={dot}
   613→   613→                      className={`inline-block h-1.5 w-5 ${
   614→   614→                        dot < item.score ? scoreBg(item.score) : "bg-muted"

   616→   616→                    />
   617→   617→                  ))}
   618→   618→                </div>
   619→   619→              </div>
   620→   620→            </div>
   621→   621→          </div>
   622→   622→        ))}
   623→   623→      </div>
   624→   624→    </div>
   625→   625→  );
   626→   626→}
