    88→function DownloadButton({ data }: { data: unknown }) {
    89→  const handleDownload = () => {
    90→    const blob = new Blob([JSON.stringify(data, null, 2)], {
    91→      type: "application/json",
    92→    });
    93→    const url = URL.createObjectURL(blob);
    94→    const a = document.createElement("a");
    95→    a.href = url;
    96→    a.download = `analysis-${Date.now()}.json`;
    97→    a.click();
    98→    URL.revokeObjectURL(url);
    99→  };
   100→  return (
   101→    <Button variant="outline" size="sm" onClick={handleDownload}>
   102→      <Download className="h-3.5 w-3.5 mr-1.5" />
   103→      Скачать JSON
   104→    </Button>
   105→  );
   106→}

   520→  }
   521→
   522→  const avg = he.averageScore ?? 0;
   523→
   524→  const scoreColor = (s: number) => {
   525→    if (s >= 3.5) return "text-emerald-600 dark:text-emerald-400";
   526→    if (s >= 2.5) return "text-amber-600 dark:text-amber-400";
   527→    if (s >= 1.5) return "text-orange-600 dark:text-orange-400";
   528→    return "text-red-600 dark:text-red-400";
   529→  };
   530→
   531→  const scoreBg = (s: number) => {
   532→    if (s >= 3.5) return "bg-emerald-500";
   533→    if (s >= 2.5) return "bg-amber-500";
   534→    if (s >= 1.5) return "bg-orange-500";
   535→    return "bg-red-500";
   536→  };
   537→
   538→  return (
   539→    <div className="space-y-6">
   540→      {/* Average score */}
   541→      <div className="border-l-2 border-l-primary/40 pl-4 py-3">
   542→        <div className="flex flex-col sm:flex-row items-center gap-4">
   543→          <div className="flex items-baseline gap-1">
   544→            <span className={`text-4xl font-bold ${scoreColor(avg)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   545→              {avg.toFixed(1)}
   546→            </span>
   547→            <span className="text-lg text-muted-foreground">/4.0</span>
   548→          </div>
   549→          <div className="text-center sm:text-left">

   555→        </div>
   556→      </div>
   557→
   558→      {/* Individual scores */}
   559→      <div className="space-y-3">
   560→        {he.scores.map((item, i) => (
   561→          <div key={i} className="border-b border-white/5 pb-3 pt-3">
   562→            <div className="flex items-start justify-between gap-4">
   563→              <div className="flex-1 space-y-1">
   564→                <p className="text-sm font-medium">{item.heuristic}</p>
   565→                {item.observations && (
   566→                  <p className="text-xs text-muted-foreground">{item.observations}</p>
   567→                )}
   568→                {item.recommendation && (
   569→                    <p className="text-xs text-muted-foreground italic mt-1">
   570→                      {item.recommendation}
   571→                    </p>
   572→                )}
   573→              </div>
   574→              <div className="flex flex-col items-center gap-1 shrink-0">
   575→                <span className={`text-lg font-bold ${scoreColor(item.score)}`} style={{ fontVariantNumeric: "tabular-nums" }}>
   576→                  {item.score}
   577→                </span>
   578→                <div className="flex gap-0.5">
   579→                  {[0, 1, 2, 3].map((dot) => (
   580→                    <span
   581→                      key={dot}
   582→                      className={`inline-block h-1.5 w-5 ${
   583→                        dot < item.score ? scoreBg(item.score) : "bg-muted"
   584→                      }`}
   585→                    />
   586→                  ))}
   587→                </div>
   588→              </div>
   589→            </div>
   590→          </div>
   591→        ))}
   592→      </div>
   593→    </div>
   594→  );
   595→}

   608→function MetaInfo({ data }: { data: AnalysisResult }) {
   609→  const m = data.meta;
   610→  if (!m) return null;
   611→
   612→  return (
   613→    <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   614→      {m.dataSources?.map((ds) => (
   615→        <Badge key={ds} variant="outline" className="text-xs">
   616→          {ds}
   617→        </Badge>
   618→      ))}
   619→      {m.confidence && <ConfidenceBadge confidence={m.confidence} />}
   620→      {m.caveats?.map((c, i) => (
   621→        <span key={i} className="text-xs text-muted-foreground italic">
   622→          {c}
   623→        </span>
   624→      ))}
   625→    </div>
   626→  );
   627→}
